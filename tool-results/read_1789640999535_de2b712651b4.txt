   247→   247→export function TelegramCard({

   255→   255→  const [token, setToken] = useState("");
   256→   256→  const [busy, setBusy] = useState(false);
   257→   257→  const [error, setError] = useState<string | null>(null);
   258→   258→  const [gateway, setGateway] = useState<{ mode?: string; warning?: string } | null>(null);
   259→   259→  const profile = useDeepInit((s) => s.profile);
   260→   260→
   261→   261→  const pair = async (mode: "builtin" | "own") => {
   262→   262→    setBusy(true);
   263→   263→    setError(null);
   264→   264→    try {
   265→   265→      const state = useDeepInit.getState();
   266→   266→      const systemPrompt = buildSystemPrompt(
   267→   267→        state.profile,
   268→   268→        state.questionnaire,
   269→   269→        state.channels,
   270→   270→        state.tools,
   271→   271→        state.instances,
   272→   272→        state.tunnels,
   273→   273→        state.skills
   274→   274→      );
   275→   275→      const res = await fetch("/api/pair/telegram", {
   276→   276→        method: "POST",
   277→   277→        headers: { "Content-Type": "application/json" },
   278→   278→        body: JSON.stringify({
   279→   279→          mode,
   280→   280→          token: mode === "own" ? token.trim() : undefined,
   281→   281→          config: {
   282→   282→            agentName: state.profile.agentName,
   283→   283→            ownerName: state.profile.displayName,
   284→   284→            ownerToken: state.profile.pairingToken,
   285→   285→            systemPrompt,
   286→   286→            providers: [...state.providers]
   287→   287→              .filter((p) => p.enabled && p.baseUrl && p.model)
   288→   288→              .sort((a, b) => a.priority - b.priority)
   289→   289→              .map((p) => ({
   290→   290→                id: p.id,
   291→   291→                label: p.label,
   292→   292→                baseUrl: p.baseUrl,
   293→   293→                apiKey: p.apiKey,

   312→   312→          sessionKey: data.sessionKey,
   313→   313→          gatewayMode: data.gatewayMode,
   314→   314→          builtIn: mode === "builtin",
   315→   315→        });
   316→   316→      } else {
   317→   317→        setError(data.error || "Pairing failed");
   318→   318→      }
   319→   319→    } catch {
   320→   320→      setError("Network error while reaching the pairing service");

   322→   322→      setBusy(false);
   323→   323→    }
   324→   324→  };

   331→   331→          <div className="flex items-center gap-2.5">
   332→   332→            <Send className="h-5 w-5 text-primary" />
   333→   333→            <span className="font-semibold">Telegram</span>
   334→   334→          </div>
   335→   335→          <div className="flex items-center gap-2">
   336→   336→            {channel.gatewayMode && (
   337→   337→              <span className="rounded-full border border-primary/40 bg-primary/10 px-2 py-0.5 font-mono text-[10px] text-primary">
   338→   338→                {channel.gatewayMode} gateway
   339→   339→              </span>
   340→   340→            )}
   341→   341→            <StatusDot ok />
   342→   342→          </div>

   345→   345→          <ChannelLine c={channel} />

   393→   393→        >

   456→   456→function StepProviders({ onNext, onBack }: { onNext: () => void; onBack: () => void }) {
   457→   457→  const providers = useDeepInit((s) => s.providers);

   521→   521→        logActivity({ kind: "provider", title: `Health check OK: ${p.label}`, detail: `${data.latencyMs}ms` });
